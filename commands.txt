# Format the EFI boot partition
mkfs.fat -F 32 -n boot /dev/nvme0n1p6

# Format the swap partition
mkswap -L swap /dev/nvme0n1p7
swapon /dev/nvme0n1p7


# Initialize LUKS encryption on the btrfs partition
cryptsetup luksFormat /dev/nvme0n1p8

# Open the encrypted container as 'cryptroot'
cryptsetup open /dev/nvme0n1p8 cryptroot


# Format the unencrypted mapper target with Btrfs
mkfs.btrfs -L nixos /dev/mapper/cryptroot

# Mount the root of the filesystem temporarily to create subvolumes
mount /dev/mapper/cryptroot /mnt

# Create the requested subvolumes
btrfs subvolume create /mnt/root
btrfs subvolume create /mnt/home
btrfs subvolume create /mnt/nix
btrfs subvolume create /mnt/data
btrfs subvolume create /mnt/log
btrfs subvolume create /mnt/snapshot

# Unmount the temporary root mount
umount /mnt


# Mount the root subvolume
mount -o subvol=root,compress=zstd /dev/mapper/cryptroot /mnt

# Create directories for the remaining mount points
mkdir -p /mnt/{home,nix,data,var/log,snapshot,boot}

# Mount the rest of the subvolumes
mount -o subvol=home,compress=zstd /dev/mapper/cryptroot /mnt/home
mount -o subvol=nix,compress=zstd /dev/mapper/cryptroot /mnt/nix
mount -o subvol=data,compress=zstd /dev/mapper/cryptroot /mnt/data
mount -o subvol=log,compress=zstd /dev/mapper/cryptroot /mnt/var/log
mount -o subvol=snapshot,compress=zstd /dev/mapper/cryptroot /mnt/snapshot

# Mount the boot partition
mount /dev/nvme0n1p6 /mnt/boot


nixos-generate-config --root /mnt

sudo nix --experimental-features "nix-command flakes" run github:nix-community/disko/latest -- --mode destroy,format,mount ./disko-config.nix
sudo nix --extra-experimental-features "nix-command flakes" nixos-install --flake .#laptop
